\noindent\textbf{Fixed and variable sugar contacts at the subsites of starch-degrading enzymes: a structural basis for processing functionalised starch}

\section*{Highlights}
\begin{itemize}
  \item Census of 637 PDB entries of 14 starch-active families maps sugar contacts by subsite
  \item 11 of 13 side-chain hydrogen-bonders at subsites $-1$/$+1$ are fixed in every lineage
  \item Aromatic residues keep an aromatic ring; aliphatic residues change between lineages
  \item Residues of every kind at the outer subsites vary between lineages
  \item Hydrogen-bonding positions exceed aliphatic ones in conservation by 0.33 (p = 0.003)
\end{itemize}
